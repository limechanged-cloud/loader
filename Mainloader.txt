--[[ aurora loader ]]

-- EDIT: your core repo
local CORE = "https://raw.githubusercontent.com/YOURNAME/aurora-core/main/core.lua"

-- EDIT: one line per tab, in the order the tabs should appear
local TABS = {
    { name = "players", url = "https://raw.githubusercontent.com/YOURNAME/aurora-players/main/players.lua" },
    { name = "esp",     url = "https://github.com/limechanged-cloud/loader/blob/main/esp" },
    { name = "combat",  url = "https://raw.githubusercontent.com/YOURNAME/aurora-combat/main/combat.lua" },
    { name = "misc",    url = "https://raw.githubusercontent.com/YOURNAME/aurora-misc/main/misc.lua" },
}

local function fetch(url)
    local chunk = game:HttpGet(url)
    if chunk == "" or chunk:find("^404") then error("empty response") end
    local fn = loadstring(chunk)
    if type(fn) ~= "function" then error("chunk did not compile") end
    return fn
end

local A = { Tabs = {} }

local ok, err = pcall(function()
    fetch(CORE)(A)
end)
if not ok or not A.Library then
    return warn("[aurora] core failed: " .. tostring(err))
end

A.Window = A.Library.Window("aurora 1.0", Vector2.new(600, 462))

for _, t in ipairs(TABS) do
    local built = pcall(function()
        fetch(t.url)(A)
    end)
    if built then
        A.Tabs[#A.Tabs + 1] = t.name
    else
        warn("[aurora] '" .. t.name .. "' failed to load")
    end
end

A.Window:AddSettingsTab(function()
    if A.Unload then A.Unload() end
end)

task.spawn(function()
    task.wait(0.35)
    pcall(function()
        A.Window.LastTab = nil
        for _, tab in ipairs(A.Window.Tabs) do tab:Install() end
    end)
end)

A.Window.Watermark("aurora 1.0")
A.Window.SendNotification("Normal", ("aurora 1.0 | %d parts"):format(#A.Tabs), 4)
